import React, { useState, useEffect } from 'react';
import { GripVertical, X, Download, RefreshCw, Table } from 'lucide-react';

const SSASCubeBrowser = () => {
  const [cubes, setCubes] = useState([]);
  const [selectedCube, setSelectedCube] = useState('');
  const [dimensions, setDimensions] = useState([]);
  const [measures, setMeasures] = useState([]);
  const [rows, setRows] = useState([]);
  const [columns, setColumns] = useState([]);
  const [selectedMeasures, setSelectedMeasures] = useState([]);
  const [data, setData] = useState(null);
  const [loading, setLoading] = useState(false);
  const [draggedItem, setDraggedItem] = useState(null);

  // API base URL - adjust for your backend
  const API_BASE = 'http://localhost:5000/api';

  useEffect(() => {
    fetchCubes();
  }, []);

  useEffect(() => {
    if (selectedCube) {
      fetchCubeMetadata(selectedCube);
    }
  }, [selectedCube]);

  const fetchCubes = async () => {
    try {
      const response = await fetch(`${API_BASE}/cubes`);
      const result = await response.json();
      setCubes(result.cubes || []);
    } catch (error) {
      console.error('Error fetching cubes:', error);
    }
  };

  const fetchCubeMetadata = async (cubeName) => {
    try {
      const response = await fetch(`${API_BASE}/cube/${cubeName}/metadata`);
      const result = await response.json();
      setDimensions(result.dimensions || []);
      setMeasures(result.measures || []);
      setRows([]);
      setColumns([]);
      setSelectedMeasures([]);
      setData(null);
    } catch (error) {
      console.error('Error fetching metadata:', error);
    }
  };

  const executeQuery = async () => {
    if (!selectedCube || selectedMeasures.length === 0) {
      alert('Please select a cube and at least one measure');
      return;
    }

    setLoading(true);
    try {
      const response = await fetch(`${API_BASE}/query`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          cube: selectedCube,
          rows: rows,
          columns: columns,
          measures: selectedMeasures
        })
      });
      const result = await response.json();
      setData(result);
    } catch (error) {
      console.error('Error executing query:', error);
      alert('Error executing query');
    } finally {
      setLoading(false);
    }
  };

  const exportCSV = () => {
    if (!data || !data.data) return;

    const headers = [...(data.columnHeaders || []), ...(data.measureHeaders || [])];
    const csvContent = [
      headers.join(','),
      ...data.data.map(row => row.join(','))
    ].join('\n');

    const blob = new Blob([csvContent], { type: 'text/csv' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `${selectedCube}_export.csv`;
    a.click();
    URL.revokeObjectURL(url);
  };

  const handleDragStart = (e, item, source) => {
    setDraggedItem({ item, source });
    e.dataTransfer.effectAllowed = 'move';
  };

  const handleDrop = (e, target) => {
    e.preventDefault();
    if (!draggedItem) return;

    const { item, source } = draggedItem;

    // Remove from source
    if (source === 'rows') {
      setRows(rows.filter(r => r !== item));
    } else if (source === 'columns') {
      setColumns(columns.filter(c => c !== item));
    } else if (source === 'measures') {
      setSelectedMeasures(selectedMeasures.filter(m => m !== item));
    }

    // Add to target
    if (target === 'rows' && !rows.includes(item)) {
      setRows([...rows, item]);
    } else if (target === 'columns' && !columns.includes(item)) {
      setColumns([...columns, item]);
    } else if (target === 'measures' && !selectedMeasures.includes(item)) {
      setSelectedMeasures([...selectedMeasures, item]);
    }

    setDraggedItem(null);
  };

  const handleDragOver = (e) => {
    e.preventDefault();
    e.dataTransfer.dropEffect = 'move';
  };

  const removeFromArea = (item, area) => {
    if (area === 'rows') {
      setRows(rows.filter(r => r !== item));
    } else if (area === 'columns') {
      setColumns(columns.filter(c => c !== item));
    } else if (area === 'measures') {
      setSelectedMeasures(selectedMeasures.filter(m => m !== item));
    }
  };

  const DraggableItem = ({ item, source, onRemove }) => (
    <div
      draggable
      onDragStart={(e) => handleDragStart(e, item, source)}
      className="flex items-center gap-2 bg-blue-100 text-blue-800 px-3 py-2 rounded cursor-move hover:bg-blue-200 transition-colors"
    >
      <GripVertical size={16} />
      <span className="flex-1 text-sm">{item}</span>
      {onRemove && (
        <button onClick={() => onRemove(item)} className="hover:text-red-600">
          <X size={16} />
        </button>
      )}
    </div>
  );

  const DropZone = ({ title, items, area, emptyText }) => (
    <div className="border-2 border-dashed border-gray-300 rounded-lg p-4 min-h-32">
      <h3 className="font-semibold text-gray-700 mb-2">{title}</h3>
      <div
        onDrop={(e) => handleDrop(e, area)}
        onDragOver={handleDragOver}
        className="space-y-2 min-h-20"
      >
        {items.length === 0 ? (
          <p className="text-gray-400 text-sm italic">{emptyText}</p>
        ) : (
          items.map((item, idx) => (
            <DraggableItem
              key={idx}
              item={item}
              source={area}
              onRemove={(item) => removeFromArea(item, area)}
            />
          ))
        )}
      </div>
    </div>
  );

  return (
    <div className="min-h-screen bg-gray-50 p-6">
      <div className="max-w-7xl mx-auto">
        <h1 className="text-3xl font-bold text-gray-900 mb-6">SSAS Cube Browser</h1>

        {/* Cube Selection */}
        <div className="bg-white rounded-lg shadow p-6 mb-6">
          <label className="block text-sm font-medium text-gray-700 mb-2">
            Select Cube
          </label>
          <select
            value={selectedCube}
            onChange={(e) => setSelectedCube(e.target.value)}
            className="w-full px-4 py-2 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-blue-500"
          >
            <option value="">Choose a cube...</option>
            {cubes.map((cube, idx) => (
              <option key={idx} value={cube}>{cube}</option>
            ))}
          </select>
        </div>

        {selectedCube && (
          <div className="grid grid-cols-1 lg:grid-cols-3 gap-6 mb-6">
            {/* Available Dimensions */}
            <div className="bg-white rounded-lg shadow p-6">
              <h2 className="text-lg font-semibold text-gray-800 mb-4">Dimensions</h2>
              <div className="space-y-2 max-h-96 overflow-y-auto">
                {dimensions.map((dim, idx) => (
                  <DraggableItem key={idx} item={dim} source="dimensions" />
                ))}
              </div>
            </div>

            {/* Available Measures */}
            <div className="bg-white rounded-lg shadow p-6">
              <h2 className="text-lg font-semibold text-gray-800 mb-4">Measures</h2>
              <div className="space-y-2 max-h-96 overflow-y-auto">
                {measures.map((measure, idx) => (
                  <DraggableItem key={idx} item={measure} source="measures" />
                ))}
              </div>
            </div>

            {/* Query Builder */}
            <div className="bg-white rounded-lg shadow p-6 space-y-4">
              <h2 className="text-lg font-semibold text-gray-800 mb-4">Query Builder</h2>
              
              <DropZone
                title="Rows"
                items={rows}
                area="rows"
                emptyText="Drag dimensions here"
              />

              <DropZone
                title="Columns"
                items={columns}
                area="columns"
                emptyText="Drag dimensions here"
              />

              <DropZone
                title="Selected Measures"
                items={selectedMeasures}
                area="measures"
                emptyText="Drag measures here"
              />

              <button
                onClick={executeQuery}
                disabled={loading || selectedMeasures.length === 0}
                className="w-full bg-blue-600 text-white py-2 px-4 rounded-lg hover:bg-blue-700 disabled:bg-gray-400 disabled:cursor-not-allowed flex items-center justify-center gap-2"
              >
                {loading ? (
                  <>
                    <RefreshCw size={18} className="animate-spin" />
                    Executing...
                  </>
                ) : (
                  <>
                    <Table size={18} />
                    Execute Query
                  </>
                )}
              </button>
            </div>
          </div>
        )}

        {/* Results */}
        {data && (
          <div className="bg-white rounded-lg shadow p-6">
            <div className="flex justify-between items-center mb-4">
              <h2 className="text-lg font-semibold text-gray-800">Results</h2>
              <button
                onClick={exportCSV}
                className="bg-green-600 text-white py-2 px-4 rounded-lg hover:bg-green-700 flex items-center gap-2"
              >
                <Download size={18} />
                Export CSV
              </button>
            </div>
            <div className="overflow-x-auto">
              <table className="min-w-full divide-y divide-gray-200">
                <thead className="bg-gray-50">
                  <tr>
                    {[...(data.columnHeaders || []), ...(data.measureHeaders || [])].map((header, idx) => (
                      <th key={idx} className="px-4 py-3 text-left text-xs font-medium text-gray-700 uppercase tracking-wider">
                        {header}
                      </th>
                    ))}
                  </tr>
                </thead>
                <tbody className="bg-white divide-y divide-gray-200">
                  {(data.data || []).map((row, idx) => (
                    <tr key={idx} className="hover:bg-gray-50">
                      {row.map((cell, cellIdx) => (
                        <td key={cellIdx} className="px-4 py-3 text-sm text-gray-900 whitespace-nowrap">
                          {cell}
                        </td>
                      ))}
                    </tr>
                  ))}
                </tbody>
              </table>
            </div>
          </div>
        )}
      </div>
    </div>
  );
};

export default SSASCubeBrowser;
